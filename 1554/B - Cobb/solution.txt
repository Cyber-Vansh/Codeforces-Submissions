import sys
 
input = sys.stdin.buffer.readline
 
def solve():
    n,k = map(int,input().split())
    arr = list(map(int,input().split()))
 
    ans = -float('inf')
 
    for i in range(n-1,max(n-1-2*k-1,-1),-1):
        for j in range(n-1,max(n-1-2*k-1,-1),-1):
            if i==j:
                continue
            ans = max(ans,(i+1)*(j+1)-k*(arr[i]|arr[j]))
 
    print(ans)
 
 
t = int(input())
for _ in range(t):
    solve()
