n = int(input())
arr = list(map(int,input().split()))
 
d = {}
l = []
 
for i in range(n):
    d[arr[i]] = d.get(arr[i],0)+1
    l.append(d[arr[i]])
 
d = {}
r = []
 
for i in range(n-1,-1,-1):
    d[arr[i]] = d.get(arr[i],0)+1
    r.append(d[arr[i]])
r.reverse()
 
bit = [0]*(n+1)
 
def query(idx):
    Sum = 0
    while idx>0:
        Sum += bit[idx]
        idx -= idx&(-idx)
    return Sum
 
def update(idx,val):
    while idx<=n:
        bit[idx] += val
        idx += idx&(-idx)
 
ans = 0
 
for i in range(n):
    ans += query(n)-query(r[i])
    update(l[i],1)
 
print(ans)
