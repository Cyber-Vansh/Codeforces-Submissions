import sys
 
input = sys.stdin.buffer.readline
 
def c2(a):
    return (a*(a-1))//2 
 
def solve():
    n,k = map(int,input().split())
    arr = list(map(int,input().split()))
 
    Max = (1<<k)-1
 
    curr = 0
    pre = []
 
    for i in range(n):
        pre.append(curr^arr[i])
        curr ^= arr[i]
 
    d = {}
    for i in pre:
        if i<=i^Max:
            d[i] = d.get(i,0) + 1
        else:
            d[i^Max] = d.get(i^Max,0) + 1
 
    ans = (n*(n+1))//2
 
    for i,j in d.items():
        if i==0:
            if j%2==0:
                ans -= c2(j//2+1) + c2(j//2)
            else:
                ans -= c2(j//2+1) + c2(j//2+1)
        else:
            if j%2==0:
                ans -= c2(j//2) + c2(j//2)
            else:
                ans -= c2(j//2) + c2(j//2+1)
 
    print(ans)
 
 
 
t = 1
for _ in range(t):
    solve()
