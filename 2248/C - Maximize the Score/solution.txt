t = int(input())
for _ in range(t):
    n = int(input())
    arr = list(map(int,input().split()))
 
    left = [-1]*(n+1)
    right = [-1]*(n+1)
 
    for i in range(2*n):
        if left[arr[i]]==-1:
            left[arr[i]] = i
        else:
            right[arr[i]] = i
 
    dp = [-1]*(2*n)
    for i in range(2*n):
        if left[arr[i]]==i:
            if i>0:
                dp[i] = dp[i-1]+1
            else:
                dp[i] = 1
        else:
            x = (right[arr[i]]-left[arr[i]]+1)**2
            if left[arr[i]]>0:
                x += dp[left[arr[i]]-1]
            dp[i] = max(dp[i-1]+1,x)
 
    print(dp[2*n-1])
