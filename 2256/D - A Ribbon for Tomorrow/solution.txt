import math
 
mod = 998244353
 
fact = [1]*(10**6)
inv = [1]*(10**6)
 
for i in range(1,10**6):
    fact[i] = (fact[i-1]*i)%mod
 
inv[10**6-1] = pow(fact[10**6-1],mod-2,mod)
for i in range(10**6-2,-1,-1):
    inv[i] = (inv[i+1]*(i+1))%mod
 
def get(n,r):
    if r<0 or r>n:
        return 0
    return (fact[n]*inv[r]%mod)*(inv[n-r])%mod
 
t = int(input())
for _ in range(t):
    n = int(input())
    s = input()
 
    count0 = 0
    count1 = 0
 
    zero = 0 if s[0]=="1" else 1
    one = 0 if s[0]=="0" else 1
 
    for i in range(n):
        if s[i]=="0":
            count0 += 1
 
            if i-1>=0 and s[i]!=s[i-1]:
                zero += 1
        else:
            count1 += 1
            
            if i-1>=0 and s[i]!=s[i-1]:
                one += 1
 
    if count0==0 or count1==0:
        print(1)
        continue
 
    a = get(count0-1,zero-1)%mod
    b = get(count1-1,one-1)%mod
 
    print((a*b)%mod)
