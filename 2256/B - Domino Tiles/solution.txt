t = int(input())
for _ in range(t):
    n = int(input())
    s = list(input())
 
    flag1 = True
    check = True
 
    for i in range(0,n,2):
        if s[i]!="?":
            flag1 = False
 
        if (i!=n-1 and i!=n-2) and s[i]==s[i+2] and s[i]!="?":
            check = False
 
        if (i!=n-1 and i!=n-2) and s[i]!="?":
            if s[i]=="1":
                s[i+2] = "0"
            else:
                s[i+2] = "1"
 
 
    if not check:
        print(0)
        continue
 
    flag2 = True
 
    for i in range(1,n,2):
        if s[i]!="?":
            flag2 = False
 
        if (i!=n-1 and i!=n-2) and s[i]==s[i+2] and s[i]!="?":
            check = False
 
        if (i!=n-1 and i!=n-2) and s[i]!="?":
            if s[i]=="1":
                s[i+2] = "0"
            else:
                s[i+2] = "1"
 
    if not check:
        print(0)
        continue
 
    print((1+int(flag1))*(1+int(flag2)))
