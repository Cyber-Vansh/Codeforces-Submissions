import heapq
 
t = int(input())
for _ in range(t):
    m,x = map(int,input().split())
    c = list(map(int,input().split()))
 
    heap = []
 
    curr = 0
    ans = 0 
 
    for i in range(m):
        if curr+c[i]<=i*x:
            curr += c[i]
            heapq.heappush(heap,-c[i])
            ans += 1
        else:
            if len(heap)>0 and -heap[0]>c[i]:
                curr -= -heap[0]
                curr += c[i]
 
                heapq.heappop(heap)
                heapq.heappush(heap,-c[i])
 
    print(ans)
