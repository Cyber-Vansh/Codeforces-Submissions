t = int(input())
for _ in range(t):
    n = int(input())
 
    ans = 0
 
    for i in range(n+1):
        Max = 0
        curr = 0
 
        idx = 1
 
        for j in range(1,i+1):
            curr += j*idx
            Max = max(Max,j*idx)
            idx += 1
 
        for j in range(n,i,-1):
            curr += j*idx
            Max = max(Max,j*idx)
            idx += 1
        
        curr -= Max
 
        ans = max(ans,curr)
    
    print(ans)
