t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int,input().split()))
 
    pre = [0]
 
    for i in range(n):
        pre.append(pre[-1]+a[i])
    
    q = int(input())
    for i in range(q):
        l,u = map(int,input().split())
 
        x = pre[l-1]+u
 
        left = l
        right = n
 
        idx = l
 
        while left<=right:
            mid = (left+right)//2
 
            if pre[mid]<=x:
                idx = mid
                left = mid+1
            else:
                right = mid-1
        
        if idx==n or x-pre[idx]<pre[idx+1]-x:
            print(idx,end=" ")
        else:
            print(idx+1,end=" ")
    
    print()
