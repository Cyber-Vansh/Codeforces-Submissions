import sys
 
input = sys.stdin.buffer.readline
 
def solve():
    n,k,q = map(int,input().split())
 
    arr = [0]*200001
 
    for i in range(n):
        l,r = map(int,input().split())
 
        arr[l] += 1
        if r+1<len(arr):
            arr[r+1] -= 1
 
    for i in range(1,len(arr)):
        arr[i] += arr[i-1]
 
    valid = [1 if i>=k else 0 for i in arr]
 
    for i in range(1,len(arr)):
        valid[i] += valid[i-1]
 
    for i in range(q):
        a,b = map(int,input().split())
 
        print(valid[b]-valid[a-1])
    
    
t = 1
for _ in range(t):
    solve()
