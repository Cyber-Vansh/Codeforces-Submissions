import math
 
t = int(input())
for _ in range(t):
    n,k = map(int,input().split())
 
    if n-2<k:
        print(-1)
        continue
    
    s = ['0']*(math.ceil(k/2)+1) + ['1']*(math.floor(k/2)+1)
 
    if len(s)<n and k%2==1:
        s = ['1'] + s
 
    while len(s)<n:
        if len(s)==0 or s[-1]=='1':
            s.append('0')
        else:
            s.append('1')
    
    print("".join(s))
