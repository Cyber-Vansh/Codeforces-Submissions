t = int(input())
for _ in range(t):
    n = int(input())
    
    nxt = [1]*(n+1)
    
    for i in range(n):
        l,r,u,v = map(int,input().split())
 
        for j in range(1,n+1):
            if j<nxt[j]:
                continue
 
            # print(i,j,nxt)
 
            a = nxt[j]
            b = j-nxt[j]+1
 
            if l<=a<=r or u<=b<=v:
                continue
            
            nxt[j] += 1
            # print(i,j,nxt)
            # print()
    
    ans = 0
    for i in range(len(nxt)):
        if nxt[i]>i:
            ans = i
    
    print(ans)
