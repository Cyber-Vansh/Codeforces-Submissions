t = int(input())
for _ in range(t):
    n = int(input())
    b = list(map(int,input().split()))
 
    dp = [0]*n
 
    for i in range(n):
        if i+b[i]<n and (i==0 or dp[i-1]==1):
            dp[i+b[i]] = 1
 
        if i-b[i]==0 or (i-b[i]-1>=0 and dp[i-b[i]-1]==1):
            dp[i] = 1
 
    if dp[n-1]:
        print("YES")
    else:
        print("NO")
