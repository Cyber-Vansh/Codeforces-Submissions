import sys
 
input = sys.stdin.buffer.readline
 
def solve():
    n,m = map(int,input().split())
    arr = list(map(int,input().split()))
 
    arr.sort()
 
    mul = [0]*(m+1)
 
    ans = [0]*(m+1)
 
    for i in range(n):
        mul[arr[i]] += 1
 
    suf = [0]*(m+2)
    for i in range(m,-1,-1):
        suf[i] = suf[i+1]+mul[i]
 
    for i in range(1,m+1):
        curr = 0
        k = 1
        
        for j in range(1,(m//i)+1):
            curr += suf[j*i]
            
            if j==(1<<k)-1:
                curr_ = curr
                
                if (1<<k)*i<len(mul):
                    curr_ += mul[(1<<k)*i]
                    
                if k<=m:
                    ans[k] = max(ans[k],curr_)
                k += 1
        
        if k<=m:
            ans[k] = max(ans[k],curr)
 
    for k in range(2,m+1):
        ans[k] = max(ans[k],ans[k-1])
 
    # print(mul)
 
    print(*(ans[1:]))
 
t = int(input())
for _ in range(t):
    solve()
