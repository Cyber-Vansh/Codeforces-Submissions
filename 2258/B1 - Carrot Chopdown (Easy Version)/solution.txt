import sys
 
input = sys.stdin.buffer.readline
 
def solve():
    n,m = map(int,input().split())
    arr = list(map(int,input().split()))
 
    arr.sort()
 
    mul = [0]*(m+1)
 
    ans = 0
 
    for i in range(n):
        mul[arr[i]] += 1
 
    prev = 0
    for i in range(len(mul)-1,-1,-1):
        if i*2<len(mul):
            ans = max(ans,prev+mul[i]+mul[i*2])
        else:
            ans = max(ans,prev+mul[i])
 
        prev += mul[i]
 
    # print(mul)
 
    print(ans)
 
 
t = int(input())
for _ in range(t):
    solve()
