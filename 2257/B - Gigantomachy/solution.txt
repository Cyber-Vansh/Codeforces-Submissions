import sys
 
input = sys.stdin.buffer.readline
 
def solve():
    n,m = map(int,input().split())
    a = list(map(int,input().split()))
    b = list(map(int,input().split()))
    
    bea = a[0]+n-1
    ver = b[0]+m-1
    
    if bea>=ver:
        print(1)
    else:
        print(2)
 
t = int(input())
for _ in range(t):
    solve()
