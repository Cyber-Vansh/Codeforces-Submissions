gpf = [i for i in range(10**5+1)]
 
for i in range(2,len(gpf)): 
    if gpf[i]==i:
        for j in range(i*2,len(gpf),i):
            gpf[j] = i
 
t = int(input())
for _ in range(t):
    n = int(input())
 
    arr = [[] for i in range(n+1)]
    for i in range(1,n+1):
        arr[gpf[i]].append(i)
 
    for i in range(n+1):
        if len(arr[i])>0:
            arr[i] = arr[i][1:]+[arr[i][0]]
            arr[i] = arr[i][::-1]
 
 
    for i in range(1,n+1):
        print(arr[gpf[i]][-1],end=" ")
        arr[gpf[i]].pop()
 
    print()
