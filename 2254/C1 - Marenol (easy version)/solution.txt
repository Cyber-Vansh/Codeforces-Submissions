t = int(input())
for _ in range(t):
    n = int(input())
    a = input()
    b = input()
 
    evena = 0
    evenb = 0
    odda = 0
    oddb = 0
 
    for i in range(n):
        if i%2==0:
            if a[i]=="1":
                evena += 1
            if b[i]=="1":
                evenb += 1
        else:
            if a[i]=="1":
                odda += 1
            if b[i]=="1":
                oddb += 1
 
    if evena==evenb and odda==oddb:
        print("YES")
    else:
        print("NO")
