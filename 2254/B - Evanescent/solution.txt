t = int(input())
for  _ in range(t):
    n = int(input())
    s = input()
 
    flag = False
    ans = 1
 
    check = False
 
    for i in range(1,n):
        if s[i-1]==s[i]:
            continue
        if i<n-1 and s[i]!=s[i+1]:
            flag = True
 
        if i<n-1 and s[i-1]==s[i+1] and not check:
            check = True
            ans -= 1
 
        ans += 1
        
 
    if flag:
        ans -= 1
 
    print(ans)
