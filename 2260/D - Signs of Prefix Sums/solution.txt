import sys
 
input = sys.stdin.readline
 
def solve():
    n = int(input())
    s = input()
 
    if s[0]=="0" or s.count("00"):
        print(-1)
        return 
 
    check1 = True
 
    Min = 0
    Max = 0
 
    for i in range(n):
        Min -= 1
        Max += 1
 
        if s[i]=="+":
            Min = 1
        elif s[i]=="-":
            Max = -1
        else:
            Min = 0
            Max = 0
 
        if i%2==0:
            if Min%2==0:
                Min += 1
            if Max%2==0:
                Max -= 1
        else:
            if Min%2==1:
                Min += 1
            if Max%2==1:
                Max -= 1
 
        if Min>Max:
            check1 = False
            break
 
    if check1:
        print(1)
        return
 
    check2 = True
 
    Min = 0
    Max = 0
 
    for i in range(n):
        a,b = Min,Max
 
        Min -= 2
        Max += 2
 
        if s[i]=="+":
            Min = 1
        elif s[i]=="-":
            Max = -1
        else:
            Min = 0
            Max = 0
 
        if a==b:
            if Min==a:
                Min += 1
            if Max==a:
                Max -= 1
 
        if Min>Max:
            check2 = False
            break
 
    if check2:
        print(2)
        return
 
    print(3)
        
 
 
t = int(input())
for _ in range(t):
    solve()
