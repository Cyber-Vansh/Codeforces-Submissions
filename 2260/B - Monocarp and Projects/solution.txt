import sys
 
input = sys.stdin.buffer.readline
 
def solve():
    x,y,k = map(int,input().split())
 
    ans = 0
 
    while (k>0):
        if y//x==1:
            break
        
        ans += y%x
 
        x += 1
        y += 1
        k -= 1
 
    if k>0:
        ans += k*(y-x)
 
    print(ans)
 
t = int(input())
for _ in range(t):
    solve()
