import sys
 
input = sys.stdin.buffer.readline
 
def solve():
    n = int(input())
    arr = list(map(int,input().split()))
 
    valid = [1]*n
 
    for i in range(n):
        for j in range(arr[i]*(i+1),arr[i]*(i+1)+i+1):
            if j>=n:
                break
            valid[j] = 0
 
    print(valid.count(1))
    for i in range(n):
        if valid[i]:
            print(i,end=" ")
    print()
 
t = int(input())
for _ in range(t):
    solve()
