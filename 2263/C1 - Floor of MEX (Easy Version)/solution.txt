import sys
 
input = sys.stdin.buffer.readline
 
def solve():
    n = int(input())
    arr = list(map(int,input().split()))
 
    valid = [0]*n
 
    for i in range(n):
        if arr[i]*(i+1)<n:
            valid[arr[i]*(i+1)] += -1
        if arr[i]*(i+1)+i+1<n:
            valid[arr[i]*(i+1)+i+1] += 1
 
    for i in range(1,n):
        valid[i] += valid[i-1]
 
    print(valid.count(0))
    for i in range(n):
        if valid[i]==0:
            print(i,end=" ")
    print()
 
t = int(input())
for _ in range(t):
    solve()
